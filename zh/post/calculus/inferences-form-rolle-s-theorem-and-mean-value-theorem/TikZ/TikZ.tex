\newif\ifdimoutside
\tikzset{
	dim/outside/.is if=dimoutside,
	pics/showDistance/.style n args={5}{
		code={
			% Get start & endpoint coordinates
			\coordinate (P) at (#1);
			\coordinate (Q) at (#2);
			\path (P);
			\pgfgetlastxy{\xP}{\yP}
			\path (Q);
			\pgfgetlastxy{\xQ}{\yQ}

			% Scale multiplier
			\coordinate (Pp) at ({\xP*#3},{\yP*#3});
			\coordinate (Qp) at ({\xQ*#3},{\yQ*#3});

			% Get extending distance
			\pgfmathsetlengthmacro{\dis}{#5*#3*1cm}

			% Place node
			\node at ({(\xP+\xQ)/2*#3},{(\yP+\yQ)/2*#3}) {#4};
			
			% Draw lines with arrow
			\ifdimoutside
				\draw[-{latex}|] ($(Pp)!-\dis!(Qp)$) -- (Pp);
				\draw[-{latex}|] ($(Qp)!-\dis!(Pp)$) -- (Qp);
			\else
				\draw[-{latex}|] ($(Pp)!\dis!(Qp)$) -- (Pp);
				\draw[-{latex}|] ($(Qp)!\dis!(Pp)$) -- (Qp);
			\fi
  		}
	}
}
